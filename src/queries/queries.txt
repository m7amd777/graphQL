        query trans1 {
            transaction (

                    }
                }
                transaction_type : {
                    type: {_eq: "xp"}
                }
                }
            ){
                createdAt
                amount
                transaction_type {

}

export function skills() {
    const query = `
        query trans2 {
            transaction(

                }
            ) {

                type

                object {
